function host(v: string) {
  const [h, p] = (process.env[v] as string).split(':');
  return { host: h, port: Number(p) };
}

function as(uid: string) {
  const ctx = env.authenticatedContext(uid);
  (global as any).__compat = { firestore: ctx.firestore(), storage: ctx.storage(), auth: { currentUser: { uid } } };
  return ctx;
}

async function flow(name: string, fn: () => Promise<unknown>) {
  try {
    await fn();
    results.push({ flow: name, ok: true });
  } catch (e: any) {
    results.push({ flow: name, ok: false, error: String(e?.message ?? e).replace(/\s+/g, ' ') });
  }
}

// Abusos con un cliente modificado (no con la APK): ¿la regla los bloquea?
it('abusos', async () => {
  const db = as('alice').firestore();
  const { writeBatch } = require('firebase/firestore');
  const batch = async (fn: (b: any) => void) => {
    await env.withSecurityRulesDisabled(async (ctx) => { await updateDoc(doc(ctx.firestore(), 'users/alice'), { pointsBalance: 300 }); });
    const b = writeBatch(db); fn(b); await b.commit();
  };
  const now = serverTimestamp();
  await flow('A1 crédito falso report_verified +500', () => batch((b) => b.set(doc(db, 'pointTransactions/fake'), {
    userId: 'alice', type: 'report_verified', amount: 500, reportId: 'rPend', createdAt: now })));
  await flow('A2 subir pointsBalance a 9999', () => batch((b) => b.update(doc(db, 'users/alice'), { pointsBalance: 9999 })));
  await flow('A3 cambiar role a admin', () => batch((b) => b.update(doc(db, 'users/alice'), { role: 'admin' })));
  await flow('A4 subir totalPointsEarned', () => batch((b) => b.update(doc(db, 'users/alice'), { totalPointsEarned: 9999 })));
  await flow('A5 canje sin débito de saldo', () => batch((b) => b.set(doc(db, 'redemptions/free'), {
    userId: 'alice', rewardId: 'kit', pointsSpent: 150, status: 'pendiente', createdAt: now })));
  await flow('A6 dos canjes con un solo débito (legacy)', () => batch((b) => {
    b.update(doc(db, 'users/alice'), { pointsBalance: 150 });
    b.set(doc(db, 'redemptions/d1'), { userId: 'alice', rewardId: 'kit', pointsSpent: 150, status: 'pendiente', createdAt: now });
    b.set(doc(db, 'redemptions/d2'), { userId: 'alice', rewardId: 'kit', pointsSpent: 150, status: 'pendiente', createdAt: now });
  }));
  await flow('A7 dos canjes con un solo débito (cliente nuevo, lastRedemptionId)', () => batch((b) => {
    b.update(doc(db, 'users/alice'), { pointsBalance: 150, lastRedemptionId: 'n1' });
    b.set(doc(db, 'redemptions/n1'), { userId: 'alice', rewardId: 'kit', pointsSpent: 150, status: 'pendiente', createdAt: now });
    b.set(doc(db, 'redemptions/n2'), { userId: 'alice', rewardId: 'kit', pointsSpent: 150, status: 'pendiente', createdAt: now });
  }));
  await flow('A8 débito de canje sin canje (legacy, daño propio)', () => batch((b) => {
    b.update(doc(db, 'users/alice'), { pointsBalance: 150 });
    b.set(doc(db, 'pointTransactions/orphan'), { userId: 'alice', type: 'redemption', amount: -150, rewardId: 'kit',
      balanceBefore: 300, balanceAfter: 150, createdAt: now });
  }));
  await flow('A9 transacción legacy con monto positivo', () => batch((b) => {
    b.update(doc(db, 'users/alice'), { pointsBalance: 150 });
    b.set(doc(db, 'pointTransactions/pos'), { userId: 'alice', type: 'redemption', amount: 150, rewardId: 'kit',
      balanceBefore: 300, balanceAfter: 450, createdAt: now });
  }));
  await flow('A10 crear reporte ya verificado con puntos (SEC-01)', () => batch((b) => b.set(doc(db, 'reports/forged'), {
    userId: 'alice', category: 'pesca_ilegal', title: 'X', status: 'verificado', pointsAwarded: 100, photoURLs: [] })));
  await flow('A11 dueño cambia estado de su reporte (SEC-02)', () => batch((b) => b.update(doc(db, 'reports/rPend'), { status: 'verificado' })));
});
